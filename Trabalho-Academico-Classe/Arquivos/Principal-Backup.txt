\makechapterstyle{CapituloEstiloso}{%
  \chapterstyle{default}
  
  % --- Espaçamento Vertical Geral ---
  \setlength{\beforechapskip}{-1.1cm} 
  \setlength{\afterchapskip}{1cm} % distância entre Texto e Capitulo
  
  % Ajustado de -58pt para -21pt para compensar a linha curta. Mexa aqui para subir/descer o texto.
  \setlength{\midchapskip}{-15pt} 

  % Limpa o texto "Capítulo" padrão
  \renewcommand*{\printchaptername}{} 
  
  % Fonte do Número
  \renewcommand*{\chapnumfont}{\normalfont\bfseries\fontsize{35}{35}\selectfont\color{white}}

  % --- O Desenho da Caixa (Use 'cm' para tamanhos) ---
  \renewcommand*{\printchapternum}{%
     \noindent
     \begin{tikzpicture}
       % 1. O Retângulo Negro
        \node[fill=black!40, text=white, font=\chapnumfont,
              minimum width=2cm, minimum height=1.4cm, anchor=east, inner sep=0pt]
              (numbox) at (\linewidth,0) {
              \raisebox{-6pt}{\thechapter}
        };
       
       % 2. O texto "CAPÍTULO"
       \node[anchor=south east, color=black!100, 
             font=\itshape\fontsize{14}{14}\selectfont, 
             inner xsep=8pt, yshift=1pt] 
             at (numbox.west) {\chaptername};

       % 3. A Linha Horizontal Superior
       \draw[line width=0.4pt, color=black!100] (numbox.west) -- (0,0);

       % 4. A Linha Vertical (Curta, sem truques)
       % Define o tamanho real da linha aqui.
       \draw[line width=0.4pt, color=black!100] (0,0) -- (0, -0.8cm);
     \end{tikzpicture}%
  }

  % --- Impressão do Título ---
  \renewcommand*{\printchaptertitle}[1]{%
    \par
    \raggedright % alinha à direita
    \Huge\bfseries\scshape\linespread{1.2}\selectfont
    \textcolor{black!100}{##1}%
  }
  % 5. A Linha Inferior Horizontal
  \renewcommand*{\afterchaptertitle}{%
    \par\nobreak\vspace{-25pt} % Subir a reta horizontal inferior
    {\color{black!100}\rule{\linewidth}{1pt}}
    \par\nobreak\vskip \afterchapskip
  }
}
